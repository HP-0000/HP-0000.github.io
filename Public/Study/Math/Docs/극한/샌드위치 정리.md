[엡실론 델타 극한 정의](/Public/Study/Math/Docs/극한/엡실론%20델타%20극한%20정의)

> $a$ 부근에서 $g(x) \le f(x) \le h(x)$ 이고, $g(x)$와 $h(x)$의 극한값이 모두 $L$이면, 
> 가운데 낀 $f(x)$의 극한값도 $L$이다.


## 증명

$$\small
\begin{array}{rll}
1. & P_1 : \forall x \left( (x \neq a) \implies (g(x) \le f(x) \land f(x) \le h(x)) \right) 
   & [\text{Premise}] \\
2. & P_2 : \forall \epsilon \left( (\epsilon > 0) \implies \exists \delta_1 \left( (\delta_1 > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta_1)) \implies (|g(x) - L| < \epsilon) \right) \right) \right) 
   & [\text{Premise}] \\
3. & P_3 : \forall \epsilon \left( (\epsilon > 0) \implies \exists \delta_2 \left( (\delta_2 > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta_2)) \implies (|h(x) - L| < \epsilon) \right) \right) \right) 
   & [\text{Premise}] \\
\hline
4. & \quad (\epsilon_0 > 0) & [\text{Level 1 Scope Open: } \implies I] \\
5. & \quad (\epsilon_0 > 0) \implies \exists \delta_1 \left( (\delta_1 > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta_1)) \implies (|g(x) - L| < \epsilon_0) \right) \right) & [\forall E \text{ on 2, } \epsilon := \epsilon_0] \\
6. & \quad \exists \delta_1 \left( (\delta_1 > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta_1)) \implies (|g(x) - L| < \epsilon_0) \right) \right) 
   & [\implies E \text{ on 5, 4}] \\
7. & \quad (\epsilon_0 > 0) \implies \exists \delta_2 \left( (\delta_2 > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta_2)) \implies (|h(x) - L| < \epsilon_0) \right) \right) 
   & [\forall E \text{ on 3, } \epsilon := \epsilon_0] \\
8. & \quad \exists \delta_2 \left( (\delta_2 > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta_2)) \implies (|h(x) - L| < \epsilon_0) \right) \right) 
   & [\implies E \text{ on 7, 4}] \\
9. & \quad \quad (d_1 > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < d_1)) \implies (|g(x) - L| < \epsilon_0) \right) 
   & [\text{Level 2 Scope Open: } \exists E \text{ for } d_1] \\
10. & \quad \quad (d_1 > 0) & [\land E_1 \text{ on 9}] \\
11. & \quad \quad \forall x \left( ((0 < |x - a|) \land (|x - a| < d_1)) \implies (|g(x) - L| < \epsilon_0) \right) & [\land E_2 \text{ on 9}] \\
12. & \quad \quad \quad (d_2 > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < d_2)) \implies (|h(x) - L| < \epsilon_0) \right) & [\text{Level 3 Scope Open: } \exists E \text{ for } d_2] \\
13. & \quad \quad \quad (d_2 > 0) & [\land E_1 \text{ on 12}] \\
14. & \quad \quad \quad \forall x \left( ((0 < |x - a|) \land (|x - a| < d_2)) \implies (|h(x) - L| < \epsilon_0) \right) 
    & [\land E_2 \text{ on 12}] \\
15. & \quad \quad \quad (\min(d_1, d_2) > 0) & [\text{Real Arith on 10, 13}] \\
16. & \quad \quad \quad \quad ((0 < |x_0 - a|) \land (|x_0 - a| < \min(d_1, d_2))) 
    & [\text{Level 4 Scope Open: } \implies I] \\
17. & \quad \quad \quad \quad (0 < |x_0 - a|) & [\land E_1 \text{ on 16}] \\
18. & \quad \quad \quad \quad (|x_0 - a| < \min(d_1, d_2)) & [\land E_2 \text{ on 16}] \\
19. & \quad \quad \quad \quad (|x_0 - a| < d_1) & [\text{Real Arith on 18}] \\
20. & \quad \quad \quad \quad (|x_0 - a| < d_2) & [\text{Real Arith on 18}] \\
21. & \quad \quad \quad \quad ((0 < |x_0 - a|) \land (|x_0 - a| < d_1)) & [\land I \text{ on 17, 19}] \\
22. & \quad \quad \quad \quad ((0 < |x_0 - a|) \land (|x_0 - a| < d_2)) & [\land I \text{ on 17, 20}] \\
23. & \quad \quad \quad \quad ((0 < |x_0 - a|) \land (|x_0 - a| < d_1)) \implies (|g(x_0) - L| < \epsilon_0) & [\forall E \text{ on 11, } x := x_0] \\
24. & \quad \quad \quad \quad (|g(x_0) - L| < \epsilon_0) & [\implies E \text{ on 23, 21}] \\
25. & \quad \quad \quad \quad ((0 < |x_0 - a|) \land (|x_0 - a| < d_2)) \implies (|h(x_0) - L| < \epsilon_0) & [\forall E \text{ on 14, } x := x_0] \\
26. & \quad \quad \quad \quad (|h(x_0) - L| < \epsilon_0) & [\implies E \text{ on 25, 22}] \\
27. & \quad \quad \quad \quad (x_0 \neq a) & [\text{Real Arith on 17}] \\
28. & \quad \quad \quad \quad (x_0 \neq a) \implies (g(x_0) \le f(x_0) \land f(x_0) \le h(x_0)) & [\forall E \text{ on 1, } x := x_0] \\
29. & \quad \quad \quad \quad (g(x_0) \le f(x_0) \land f(x_0) \le h(x_0)) & [\implies E \text{ on 28, 27}] \\
30. & \quad \quad \quad \quad (L - \epsilon_0 < g(x_0)) & [\text{Real Arith on 24}] \\
31. & \quad \quad \quad \quad (h(x_0) < L + \epsilon_0) & [\text{Real Arith on 26}] \\
32. & \quad \quad \quad \quad (L - \epsilon_0 < f(x_0) \land f(x_0) < L + \epsilon_0) & [\text{Real Arith on 29, 30, 31}] \\
33. & \quad \quad \quad \quad (|f(x_0) - L| < \epsilon_0) & [\text{Real Arith on 32}]
\end{array}
$$
$$\small
\begin{array}{rll}
34. & \quad \quad \quad ((0 < |x_0 - a|) \land (|x_0 - a| < \min(d_1, d_2))) \implies (|f(x_0) - L| < \epsilon_0) & [\text{Close Level 4: } \implies I \text{ on 16--33}] \\
35. & \quad \quad \quad \forall x \left( ((0 < |x - a|) \land (|x - a| < \min(d_1, d_2))) \implies (|f(x) - L| < \epsilon_0) \right) & [\forall I \text{ on 34}] \\
36. & \quad \quad \quad (\min(d_1, d_2) > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \min(d_1, d_2))) \implies (|f(x) - L| < \epsilon_0) \right) & [\land I \text{ on 15, 35}] \\
37. & \quad \quad \quad \exists \delta \left( (\delta > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta)) \implies (|f(x) - L| < \epsilon_0) \right) \right) & [\exists I \text{ on 36, } t := \min(d_1, d_2)] \\
38. & \quad \quad \exists \delta \left( (\delta > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta)) \implies (|f(x) - L| < \epsilon_0) \right) \right) & [\text{Close Level 3: } \exists E \text{ on 8, 12--37}] \\
39. & \quad \exists \delta \left( (\delta > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta)) \implies (|f(x) - L| < \epsilon_0) \right) \right) & [\text{Close Level 2: } \exists E \text{ on 6, 9--38}] \\
40. & (\epsilon_0 > 0) \implies \exists \delta \left( (\delta > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta)) \implies (|f(x) - L| < \epsilon_0) \right) \right) & [\text{Close Level 1: } \implies I \text{ on 4--39}] \\
41. & \forall \epsilon \left( (\epsilon > 0) \implies \exists \delta \left( (\delta > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta)) \implies (|f(x) - L| < \epsilon) \right) \right) \right) & [\forall I \text{ on 40}]
\end{array}
$$

$$\therefore \{ P_1, P_2, P_3 \} \vdash \forall \epsilon \left( (\epsilon > 0) \implies \exists \delta \left( (\delta > 0) \land \forall x \left( ((0 < |x - a|) \land (|x - a| < \delta)) \implies (|f(x) - L| < \epsilon) \right) \right) \right)$$
